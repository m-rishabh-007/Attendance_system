# - tflite-runtime is NOT available for Python 3.12+ (PyPI limitation)

# - NumPy/MediaPipe/OpenCV version conflicts on Python 3.12
#

#
#   sudo add-apt-repository ppa:deadsnakes/ppa
#   sudo apt update
#   sudo apt install python3.11 python3.11-venv python3.11-dev
#   python3.11 -m venv venv_py311
#   source venv_py311/bin/activate
#   pip install -r requirements.txt
#
# Daily Usage:
#   source venv_py311/bin/activate  # Activate Python 3.11 environment
#   python attendance_system.py      # Run the system
#   deactivate                       # When done
#
