# ============================================================================
# Face Attendance System - Python 3.11 Requirements
# ============================================================================
# 
# ⚠️  CRITICAL: Python 3.11 REQUIRED
# 
# This project requires Python 3.11 because:
# - tflite-runtime is NOT available for Python 3.12+ (PyPI limitation)
# - TensorFlow 2.16+ has circular dependency issues with MediaPipe on 3.12
# - NumPy/MediaPipe/OpenCV version conflicts on Python 3.12
#
# Installation on Ubuntu 24.04 (or any Ubuntu with Python 3.12 default):
#
#   sudo add-apt-repository ppa:deadsnakes/ppa
#   sudo apt update
#   sudo apt install python3.11 python3.11-venv python3.11-dev
#   python3.11 -m venv venv_py311
#   source venv_py311/bin/activate
#   pip install -r requirements.txt
#
# Daily Usage:
#   source venv_py311/bin/activate  # Activate Python 3.11 environment
#   python attendance_system.py      # Run the system
#   deactivate                       # When done
#
# ============================================================================

# Core Dependencies - LOCKED VERSIONS (DO NOT CHANGE)
# These versions are carefully tested to work together on Python 3.11

# NumPy - MUST be <2.0 for MediaPipe compatibility
numpy>=1.23.0,<2.0

# OpenCV - 4.10.0.84 is stable with NumPy 1.x (DO NOT upgrade to 4.12+)

tflite-runtime>=2.14.0

# Ultralytics - YOLO utilities (NOT used for inference, only utilities)
ultralytics>=8.0.0

# PyYAML - Configuration file parsing
PyYAML>=6.0
